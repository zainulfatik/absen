Panduan Kode Google Apps Script

Salin kode JavaScript berikut ke menu Extensions > Apps Script pada Spreadsheet masing-masing kelas Anda, lalu deploy sebagai Web App (Akses: Anyone / Siapa Saja).

// ==========================================
// KODE GOOGLE APPS SCRIPT PRESENSI SISWA
// SDN DUKUH MENANGGAL I/424 SURABAYA
// ==========================================

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var monthName = contents.month; // Contoh: "Januari"
    var attendanceData = contents.data; // Array object [{nama: 'Ahmad', dates: {1:'•', 2:'S'}}]
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(monthName);
    
    // Buat Sheet Baru Jika Belum Ada
    if (!sheet) {
      sheet = ss.insertSheet(monthName);
      // Header Table
      var headers = ["No", "Nama Siswa"];
      for (var d = 1; d <= 31; d++) {
        headers.push(d.toString());
      }
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#dcfce7");
    }
    
    // Tulis Data Presensi Siswa
    sheet.getRange(2, 1, sheet.getLastRow() + 1, 33).clearContent();
    
    var rows = [];
    for (var i = 0; i < attendanceData.length; i++) {
      var item = attendanceData[i];
      var row = [i + 1, item.nama];
      for (var day = 1; day <= 31; day++) {
        row.push(item.dates[day] || "");
      }
      rows.push(row);
    }
    
    if (rows.length > 0) {
      sheet.getRange(2, 1, rows.length, rows[0].length).setValues(rows);
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success", 
      message: "Data presensi bulan " + monthName + " berhasil disimpan!"
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", 
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput("Service Presensi Berjalan Normal.");
}
💡 Langkah Deploy Apps Script:
  1. Buka Google Spreadsheet untuk kelas terkait.
  2. Klik Extensions > Apps Script.
  3. Hapus isi default, lalu tempel kode di atas. Simpan proyek.
  4. Klik Deploy > New deployment.
  5. Pilih Select type: Web app.
  6. Set Execute as: Me, dan Who has access: Anyone.
  7. Klik Deploy, lalu salin Web App URL dan tempel di Pengaturan Kelas pada aplikasi ini.
